let pontos = 0; let pontosPorClick = 1; let precoMelhoria1 = 10; const elementoPontos = document.querySelector("#pontos"); const elementoBotao = document.querySelector("#clickUP"); const elementoComprarMelhoria = document.querySelector("#comprarMelhoria"); function atualizarInterface(){ elementoPontos.textContent = pontos; elementoComprarMelhoria.textContent = "Comprar melhoria | preço:" + precoMelhoria1; } function aumentarPontos(){ pontos += pontosPorClick; atualizarInterface(); } function comprarMelhoria(){ if(pontos >= precoMelhoria1){ pontos -= precoMelhoria1; pontosPorClick++; atualizarInterface(); } else{ console.log("pontos insuficientes!"); } } atualizarInterface(); elementoBotao.addEventListener("click", aumentarPontos); elementoComprarMelhoria.addEventListener("click", comprarMelhoria);
